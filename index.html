Cara Menempelkan Kode ke Google Apps Script (GAS) & Menghubungkan API

  1. Buka Google Spreadsheet Anda > Klik menu Ekstensi (Extensions) > pilih Apps Script.
  2. Hapus semua kode bawaan lalu tempelkan (Paste) kode Apps Script murni di bawah ini.
  3. Jalankan fungsi setupSpreadsheet() sekali untuk otomatis membuat Sheet dan mengisi data anggota awal.
  4. Klik tombol Deploy > New deployment > Pilih jenis Web App (*Execute as: Me*, *Who has access: Anyone*).
  5. Salin Web App URL (contoh: https://script.google.com/macros/s/.../exec).
  6. Klik tombol Koneksi GAS API di pojok kanan atas aplikasi ini, lalu tempelkan URL tersebut!

Google Apps Script Pure API Code (Code.gs)

Salin kode ini ke Code.gs di Google Apps Script Anda.

// ==========================================
// PURE GOOGLE APPS SCRIPT JSON API BACKEND
// ==========================================

function doGet(e) {
  var output = { status: "success", members: [], cutiList: [] };
  try {
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    // Sheet 1: Data Anggota
    var sheetAnggota = ss.getSheetByName("Data Anggota");
    if (sheetAnggota && sheetAnggota.getLastRow() > 1) {
      var rawMembers = sheetAnggota.getRange(2, 1, sheetAnggota.getLastRow() - 1, 3).getValues();
      output.members = rawMembers.map(function(r) {
        return { store: String(r[0]), name: String(r[1]), role: String(r[2]) };
      });
    }
    
    // Sheet 2: Data Cuti
    var sheetCuti = ss.getSheetByName("Data Cuti");
    if (sheetCuti && sheetCuti.getLastRow() > 1) {
      var rawCuti = sheetCuti.getRange(2, 1, sheetCuti.getLastRow() - 1, 7).getValues();
      output.cutiList = rawCuti.map(function(r) {
        return {
          id: String(r[0]),
          store: String(r[1]),
          admin: String(r[2]),
          date: Number(r[3]),
          backup: String(r[4]),
          month: Number(r[5]),
          year: Number(r[6])
        };
      });
    }
  } catch(err) {
    output.status = "error";
    output.message = err.toString();
  }
  
  return ContentService.createTextOutput(JSON.stringify(output))
    .setMimeType(ContentService.MimeType.JSON);
}

function doPost(e) {
  var output = { status: "error", message: "Invalid action" };
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    if (data.action === "saveCuti") {
      var sheetCuti = ss.getSheetByName("Data Cuti");
      if (!sheetCuti) {
        sheetCuti = ss.insertSheet("Data Cuti");
        sheetCuti.appendRow(["ID", "Toko", "Admin Cuti", "Tanggal Cuti", "Admin Backup", "Bulan", "Tahun"]);
      }
      var id = "CUTI_" + new Date().getTime();
      sheetCuti.appendRow([id, data.store, data.admin, Number(data.date), data.backup, Number(data.month), Number(data.year)]);
      output = { status: "success", message: "Cuti berhasil disimpan", id: id };
    } 
    else if (data.action === "deleteCuti") {
      var sheetCuti = ss.getSheetByName("Data Cuti");
      if (sheetCuti && sheetCuti.getLastRow() > 1) {
        var rangeData = sheetCuti.getRange(2, 1, sheetCuti.getLastRow() - 1, 7).getValues();
        for (var i = 0; i < rangeData.length; i++) {
          if (String(rangeData[i][0]) === String(data.id)) {
            sheetCuti.deleteRow(i + 2);
            break;
          }
        }
      }
      output = { status: "success", message: "Cuti berhasil dihapus" };
    }
  } catch(err) {
    output = { status: "error", message: err.toString() };
  }
  
  return ContentService.createTextOutput(JSON.stringify(output))
    .setMimeType(ContentService.MimeType.JSON);
}

function setupSpreadsheet() {
  var ss = SpreadsheetApp.getActiveSpreadsheet();
  
  var sheetAnggota = ss.getSheetByName("Data Anggota");
  if (!sheetAnggota) {
    sheetAnggota = ss.insertSheet("Data Anggota");
    sheetAnggota.appendRow(["Toko", "Nama Anggota", "Role"]);
    var memberData = [
      ["AD", "Fir", "Admin"],
      ["AD", "Ist", "Admin"],
      ["CC", "Sil", "Admin"],
      ["CC", "Dev", "Admin"],
      ["CC", "Fah", "Admin"],
      ["WI", "Int", "Admin"],
      ["WI", "Nay", "Admin"],
      ["AD", "Ain", "Editor"],
      ["CC", "Nov", "Editor"],
      ["WI", "Bud", "Editor"]
    ];
    sheetAnggota.getRange(2, 1, memberData.length, 3).setValues(memberData);
  }
  
  var sheetCuti = ss.getSheetByName("Data Cuti");
  if (!sheetCuti) {
    sheetCuti = ss.insertSheet("Data Cuti");
    sheetCuti.appendRow(["ID", "Toko", "Admin Cuti", "Tanggal Cuti", "Admin Backup", "Bulan", "Tahun"]);
  }
}
